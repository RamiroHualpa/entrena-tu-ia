// 3. Para adivinar, busca los ejemplos más parecidos
function adivinar(puntosNuevos) {
  const cercanos = ejemplos
    .map(e => ({ gesto: e.gesto, dist: distancia(e.puntos, puntosNuevos) }))
    .sort((a, b) => a.dist - b.dist)
    .slice(0, 7);
  return elMasVotado(cercanos);   // ¡eso es todo!
}
